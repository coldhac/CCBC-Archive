			ctx.save();

			ctx.strokeStyle = "rgba(0, 120, 0, 1)";
			if (ctx2) ctx2.strokeStyle = "rgba(180, 255, 180, 1)";
			ctx.lineWidth = 1;
			// move to center
			ctx.translate(x, y);
			ctx.beginPath();
			ctx.arc(0, 0, 2, 0, 2 * Math.PI);
			ctx.fillStyle = "red";
			ctx.fill();

			if (ctx2) {
				ctx2.save();
				if (clearCtx2) {
					ctx2.clearRect(0, 0, 300, 300); // clear canvas
					ctx2.beginPath();
				}
				ctx2.translate(x, y);
			}

			for (let i in drawItems) {
				drawVectorAndTranslate(ctx, ctx2, drawItems[i], time,  i == drawItems.length - 1);

			}

				if (dragDir === 0) {
					removeItem(leftCadiates, dragging);
				} else if (dragDir === 1) {
					removeItem(rightCadiates, dragging);
				}

				//插入目标一侧
				if (target === 0) {
					leftCadiates.value.push(dragging);
				} else {
					rightCadiates.value.push(dragging);
				}

				//保存到本地存档中
				let saveData = {};
				let saveDataString = localStorage.getItem(LOCAL_STORAGE_SAVE_KEY);
				if (saveDataString) {
					saveData = JSON.parse(saveDataString);
				}
				if (target == 0) {
					if (saveData[currentPage.value]) {
						let ii = saveData[currentPage.value].indexOf(dragging.label);
						if (ii !== -1) {
							saveData[currentPage.value].splice(ii, 1);
						}
					}
				} else {
					if (!saveData[currentPage.value]) {
						saveData[currentPage.value] = [];
					}
					saveData[currentPage.value].push(dragging.label);
				}
				localStorage.setItem(LOCAL_STORAGE_SAVE_KEY, JSON.stringify(saveData));
			}

			dragging = null;
			dragDir = -1;
			newPath = true;
		}

            pages,
            currentPage,

			leftCadiates,
			rightCadiates,
			drag,

		}
	}

将每组 2n 个词按照某种规则分成数量相同的两组后，曲线会画出 `PVGLHRICNJFKetc8`。根据OOO OOO OOO OOO ooo (0)提示的分段，分为 `PVG LHR ICN JFK etc (8)`。前面若干三字母都是机场代号，所以本题答案是 8 个字母的 `AIRPORTS`。

## 提示

### 1. 1不会做

每个词都有一个含有数字一或者九的别名。

### 2. 2不会做

按照“是否是球形”区分。

### 3. 3不会做

每个词拼音都是英文词或者德文词。

### 4. 4不会做

每个东西都跟顺时针或者逆时针有关。

### 5. 5不会做

有些东西只有一个方向，有些东西没有一个明确方向（或者说有无数个方向）。

### 6. 6不会做

有些需要摇晃，有些则恰恰不应摇晃。

### 7. 7不会做

每个字可以跟“大”或者“小”组成常见词汇。

### 8. 8不会做

每个ABCD的词都可以缩写成AC或者BD。

### 9. 解出来了一些字母，不知道如何理解

看题目开头的文字，分段后不一定是单词，但是前四个都是属于同一类型的东西，回答该类型即可（注意多于一个，所以答案是复数）。数字代表答案的长度。
